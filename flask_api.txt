
from flask import Flask, jsonify

app = Flask(__name__)

@app.route('/')
def home():
    return jsonify({"message": "Flask API is running"})

@app.route('/student')
def student():
    return jsonify({
        "id": 1,
        "name": "Harshita",
        "course": "Computer Science"
    })

if __name__ == '__main__':
    app.run(host='0.0.0.0', port=5000)
